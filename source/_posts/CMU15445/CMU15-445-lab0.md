```cpp
#include <string>
#include <unordered_map>
using namespace std;

class Trie {
private:
    // Trie 节点结构
    struct TrieNode {
        unordered_map<char, TrieNode*> children; // 存储子节点的映射
        bool isEndOfWord = false; // 标记是否为单词结尾
    };

    TrieNode* root; // 根节点

public:
    /** 初始化 Trie 树结构 */
    Trie() {
        root = new TrieNode(); // 初始化根节点
    }

    /** 插入一个单词到 Trie 中 */
    void insert(string word) {
        TrieNode* node = root;
        for (char c : word) {
            // 如果当前字符不存在子节点，则创建一个新节点
            if (!node->children.count(c)) {
                node->children[c] = new TrieNode();
            }
            // 移动到下一个节点
            node = node->children[c];
        }
        // 标记单词结尾
        node->isEndOfWord = true;
    }

    /** 判断单词是否存在于 Trie 中 */
    bool search(string word) {
        TrieNode* node = root;
        for (char c : word) {
            // 如果某个字符不存在对应的子节点，则返回 false
            if (!node->children.count(c)) {
                return false;
            }
            // 移动到下一个节点
            node = node->children[c];
        }
        // 检查是否为单词的结尾
        return node->isEndOfWord;
    }

    /** 判断 Trie 中是否存在以某个前缀开头的单词 */
    bool startsWith(string prefix) {
        TrieNode* node = root;
        for (char c : prefix) {
            // 如果某个字符不存在对应的子节点，则返回 false
            if (!node->children.count(c)) {
                return false;
            }
            // 移动到下一个节点
            node = node->children[c];
        }
        // 找到所有前缀字符则返回 true
        return true;
    }
};
```

 `dynamic_cast` 是 C++ 中的一种类型转换操作符，主要用于在继承关系中安全地将基类指针或引用转换为派生类指针或引用。与 `static_cast` 不同，`dynamic_cast` 会在运行时进行类型检查，以确保类型转换的安全性。如果类型转换不合法，`dynamic_cast` 会返回 `nullptr`（对于指针）或抛出 `std::bad_cast` 异常（对于引用）。

### 使用场景
`dynamic_cast` 通常用于多态的继承体系中，特别是当你要将一个基类指针或引用转换为派生类类型时，并且不确定转换是否总是成功。

### 使用示例
假设你有一个基类 `Base` 和一个派生类 `Derived`：

```cpp
class Base {
  virtual ~Base() = default;  // 多态需要虚析构函数
};

class Derived : public Base {
 public:
  void Print() { std::cout << "Derived class\n"; }
};

Base* base = new Derived();

// 使用 dynamic_cast 将 base 转换为 Derived*
Derived* derived = dynamic_cast<Derived*>(base);

if (derived != nullptr) {
  derived->Print();  // 如果转换成功，可以调用 Derived 的方法
} else {
  std::cout << "转换失败\n";
}
```

### 主要特点：
1. **用于多态类型：** `dynamic_cast` 只能用于包含虚函数的类（即有多态行为的类），比如含有虚析构函数的类。
2. **运行时类型检查：** 与 `static_cast` 不同，`dynamic_cast` 会在运行时检查转换是否安全。如果不安全，它会返回 `nullptr`（对于指针）或抛出异常（对于引用）。
3. **常用于向下转换（downcasting）：** 即将基类指针或引用转换为派生类类型。

### `dynamic_cast` 的应用场景
在字典树（`Trie`）的实现中，你的树节点（`TrieNode`）可能是一个基类，而某些节点可能是带有值的派生类（如 `TrieNodeWithValue<T>`）。当你在 `Get` 方法中找到一个节点时，你不确定这个节点是否包含值，因此你可以使用 `dynamic_cast` 来检查该节点是否是 `TrieNodeWithValue<T>` 类型的实例。

在你的场景中，`dynamic_cast` 会尝试将基类 `TrieNode` 的指针转换为 `TrieNodeWithValue<T>` 的指针。如果转换成功，说明该节点确实是一个存储了值的节点，且值的类型匹配；如果转换失败，说明该节点要么没有值，要么值的类型不匹配。

### 示例：如何在你的代码中使用 `dynamic_cast`
```cpp
template <class T>
auto Trie::Get(std::string_view key) const -> const T * {
    std::shared_ptr<const TrieNode> current_node = root_;
    for (char ch : key) {
        if (!current_node || current_node->children_.find(ch) == current_node->children_.end()) {
            return nullptr;  // 键不存在
        }
        current_node = current_node->children_.at(ch);  // 移动到下一个节点
    }

    // 尝试将节点转换为包含值的节点
    auto value_node = dynamic_cast<const TrieNodeWithValue<T>*>(current_node.get());
    if (value_node == nullptr) {
        return nullptr;  // 类型不匹配或节点没有值
    }

    // 返回值的指针
    return value_node->value_.get();
}
```

在这个例子中，`dynamic_cast` 尝试将 `current_node` 转换为 `TrieNodeWithValue<T>`，如果转换成功，我们就可以访问节点中的值。如果失败，则返回 `nullptr`。

### 1. 创建 `shared_ptr` 来存储 `value`
```cpp
auto new_value = std::make_shared<T>(std::move(value));
```
此行代码的作用是：**创建一个智能指针 `new_value`**，用于安全管理 `value` 的生命周期，支持移动语义。这样做的原因是，`T` 类型的 `value` 可能不可拷贝（比如一个大型对象），用 `std::move` 可以避免不必要的复制。

### 2. 遍历字典树（Trie）并初始化子节点映射
```cpp
std::shared_ptr<const TrieNode> current_node = root_;
std::map<char, std::shared_ptr<const TrieNode>> new_children;
```
- `current_node` 指向 `Trie` 树的根节点 `root_`，表示从根节点开始遍历。
- `new_children` 是一个字符到 `TrieNode` 智能指针的映射，用于存储当前节点的子节点。当我们在遍历过程中遇到新的字符路径时，`new_children` 将会更新来表示一个新的分支。

### 3. 遍历 `key`，构建新的子树
```cpp
for (auto it = key.begin(); it != key.end(); ++it) {
    char ch = *it;
```
- 循环遍历 `key` 中的每个字符 `ch`。对于每个字符，我们沿着 `Trie` 树的路径向下查找或创建相应的节点。

### 4. 复制 `current_node` 的子节点映射，保证不可变性
```cpp
if (current_node) {
    new_children = current_node->children_;  // 复制当前节点的子节点
}
```
在 `Trie` 中，不可变性要求**不能修改原始节点**。因此，对于当前节点的 `children_`，我们先将它复制到 `new_children` 中，以便对新映射进行修改而不影响原来的树结构。

### 5. 更新或创建新的节点
- **判断是否是最后一个字符**：当到达 `key` 的最后一个字符时，我们创建一个带有 `value` 的 `TrieNodeWithValue` 节点来存储该值。
- **递归构建子节点映射**：否则，我们会沿着 `Trie` 树继续向下构建子节点。

```cpp
if (it == key.end() - 1) {  // 如果是最后一个字符

    new_children[ch] = std::make_shared<TrieNodeWithValue<T>>(std::move(new_children), new_value);

```

- **如果到达最后一个字符**，就创建一个 `TrieNodeWithValue<T>` 节点，并传入 `new_value` 和当前的 `new_children`（相当于构造了一个包含新值的新节点）。
- **否则**，我们沿着当前路径向下延伸，尝试找到或创建一个新的普通节点（`TrieNode`），并在 `new_children` 中更新该节点，以逐层构建新的 `Trie` 树结构。

### 6. 返回新的 `Trie` 根节点
```cpp
return Trie(std::make_shared<TrieNode>(std::move(new_children)));
```
- 遍历结束后，`new_children` 包含了从根节点到 `key` 的新路径。
- 将 `new_children` 传入 `TrieNode` 的构造函数生成新的根节点，实现了 `Trie` 树的不可变性。

### 总结
`Trie::Put` 方法递归遍历 `key`，沿路径创建或更新节点，最终构造出包含 `value` 的新 `Trie`，并保持了原始树结构不变。这种不可变设计确保了对同一 `Trie` 对象的任何 `Put` 操作都不会影响其他共享的树结构。
```cpp
template <class T>
auto Trie::Put(std::string_view key, T value) const -> Trie {
  // 将值存入一个 shared_ptr，以支持移动语义
  auto new_value = std::make_shared<T>(std::move(value));

  // 从根节点开始遍历
  std::shared_ptr<const TrieNode> current_node = root_;
  std::map<char, std::shared_ptr<const TrieNode>> new_children;

  // 递归地创建新子树并更新节点
  for (auto it = key.begin(); it != key.end(); ++it) {
    char ch = *it;

    // 复制当前节点的子节点映射
    if (current_node) {
      new_children = current_node->children_;  // 复制当前节点的子节点
    }

    if (it == key.end() - 1) {  // 如果是最后一个字符
      // 创建带值的新节点
      new_children[ch] = std::make_shared<TrieNodeWithValue<T>>(std::move(new_children), new_value);
    } else {
      // 创建新的 TrieNode，并递归向下
      auto next_node = current_node && current_node->children_.count(ch) 
                       ? current_node->children_.at(ch) 
                       : std::make_shared<TrieNode>();
      new_children[ch] = next_node;
      current_node = next_node;  // 移动到下一个节点
    }
  }

  // 返回新的 Trie，其中 root_ 是新构造的根节点
  return Trie(std::make_shared<TrieNode>(std::move(new_children)));
}
```
